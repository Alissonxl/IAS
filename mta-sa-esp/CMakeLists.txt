    message(FATAL_ERROR "Use Windows x86: cmake -S . -B build -A Win32")

add_executable(mta-sa-hook-host tests/hook_host.cpp)
add_library(mta-sa-hook-probe SHARED tests/hook_probe.cpp)
add_executable(mta-sa-hook-checks tests/hook_checks.cpp)
add_dependencies(mta-sa-hook-checks mta-sa-hook-host mta-sa-hook-probe)
set_target_properties(mta-sa-hook-probe PROPERTIES PREFIX "")
target_link_libraries(mta-sa-loader PRIVATE user32)
target_link_libraries(mta-sa-hook-host PRIVATE user32)
target_link_libraries(mta-sa-hook-probe PRIVATE user32)
target_link_libraries(mta-sa-hook-checks PRIVATE user32)
target_link_libraries(mta-sa-render-checks PRIVATE d3d9 user32 gdi32)
foreach(target mta-sa-esp mta-sa-loader mta-sa-checks mta-sa-render-checks mta-sa-overlay-checks mta-sa-panel-checks mta-sa-hook-host mta-sa-hook-probe mta-sa-hook-checks mta-sa-motion-checks mta-sa-panel-input-checks mta-sa-cursor-lease-checks mta-sa-aim-checks mta-sa-aim-panel-checks)
    target_include_directories(${target} PRIVATE src)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        target_link_options(${target} PRIVATE -static)
    endif()
endforeach()
